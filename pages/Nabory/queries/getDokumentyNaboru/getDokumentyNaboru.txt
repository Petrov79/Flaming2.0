SELECT
    id,
    split_part(name, '/', 2) AS name,
    created_at,
    ROUND((metadata->>'size')::numeric / 1024, 1) AS size,
    metadata->>'mimetype' AS mimetype
FROM storage.objects
WHERE bucket_id = 'dokumentacja'
  AND name LIKE CONCAT({{ appsmith.store.wybranyNaborId }}, '/%')
  AND split_part(name, '/', 2) <> '.keep'
ORDER BY name;
