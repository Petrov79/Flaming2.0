SELECT
    x.kod_gminy,
    LEFT(x.kod_gminy, 4) AS kod_powiatu,
    x.tytul,
    NULLIF(x.data_zatwierdzenia, '')::date AS data_zatwierdzenia,
    NULLIF(x.data_wykazu, '')::date AS data_wykazu,
    NULLIF(x.link, '') AS link
FROM jsonb_to_recordset(
    '{{ JSON.stringify(ImportGPR.data) }}'::jsonb
) AS x(
    kod_gminy text,
    tytul text,
    data_zatwierdzenia text,
    data_wykazu text,
    link text
)
ORDER BY
    x.kod_gminy;
