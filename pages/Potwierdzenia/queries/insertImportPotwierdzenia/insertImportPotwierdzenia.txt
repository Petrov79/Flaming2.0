INSERT INTO potwierdzenia (
    ekspert_id,
    nabor_id,
    nr_wniosku,
    data_oceny,
    rodzaj_oceny,
    kwota,
    pracownik_id
)
SELECT
    ekspert_id,
    nabor_id,
    nr_wniosku,
    data_oceny,
    rodzaj_oceny,
    kwota,
    pracownik_id
FROM jsonb_to_recordset(
    {{ JSON.stringify(appsmith.store.importPotwierdzenia || []) }}::jsonb
) AS dane (
    ekspert_id integer,
    nabor_id integer,
    nr_wniosku text,
    data_oceny date,
    rodzaj_oceny text,
    kwota numeric,
    pracownik_id integer
)
WHERE EXISTS (
    SELECT 1
    FROM sessions
    WHERE token = {{ appsmith.store.session?.token || null }}::uuid
      AND expires_at > NOW()
)
RETURNING id;
